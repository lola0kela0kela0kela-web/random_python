        sum = sum + số

"""2) Viết một hàm kiểm tra xem một số nhập từ người dùng có phải là số nguyên tố hay không."""
import math

đầu_vào = int(input("Nhập: "))
số_nguyên_tố = True

def check_số_nguyên_tố():
    global số_nguyên_tố
    for i in range(2, int(math.sqrt(đầu_vào)) + 1):
        if đầu_vào % i == 0:
            số_nguyên_tố = False

    return số_nguyên_tố

s = check_số_nguyên_tố()

if số_nguyên_tố == True:
    print(f"{đầu_vào} là một số nguyên tố")
elif số_nguyên_tố == False:
    print(f"{đầu_vào} không phải là một số nguyên tố")

đầu_vào = số
giai_thừa = [số]
kết_quả = 1
for i in range(số - 1):
    số = số - 1
    giai_thừa.append(số)

for i in giai_thừa:
    kết_quả = kết_quả * i
