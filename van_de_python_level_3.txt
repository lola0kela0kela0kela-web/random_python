"""2) Viết một chương trình tính tổng n số đầu tiên trong dãy Fibonacci."""

n = int(input("Tính tổng bao nhiêu số?: "))
fibonacci = [0, 1]
thứ_tự_1 = 0
thứ_tự_2 = 1
for i in range(n):
    fibonacci.append(fibonacci[thứ_tự_1] + fibonacci[thứ_tự_2])
    thứ_tự_1 += 1
    thứ_tự_2 += 1

sum_fibonacci = sum(fibonacci) - fibonacci[-1] - fibonacci[-2]
print(sum_fibonacci)

"""3) Viết một chương trình để kiểm tra xem một chuỗi có phải là chuỗi đối xứng hay không."""
